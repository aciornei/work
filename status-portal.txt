curl -s 'http://127.0.0.1:8081/fpm-status?json&full' |
jq -r '
.processes[]
| select(.state != "Idle")
| [
    .pid,
    ."request duration",
    ."request method",
    ."request uri",
    .script
  ]
| @tsv
' |
sort -k2,2nr |
head -40

sudo nano /etc/nginx/sites-available/phpfpm-status.conf




server {
    listen 80;
    server_name phpfpm-status.domeniu.local;

    access_log /var/log/nginx/phpfpm-status-access.log;
    error_log  /var/log/nginx/phpfpm-status-error.log;

    location = /fpm-status {
        # Prometheus și acces local
        allow 127.0.0.1;
        allow IP_PROMETHEUS;
        deny all;

        include fastcgi_params;

        fastcgi_param SCRIPT_NAME     /fpm-status;
        fastcgi_param SCRIPT_FILENAME /fpm-status;

        fastcgi_pass unix:/run/php/php8.3-fpm.sock;
    }

    location / {
        return 404;
    }
}


sudo ln -s /etc/nginx/sites-available/phpfpm-status.conf \
    /etc/nginx/sites-enabled/phpfpm-status.conf
	
	
	sudo nginx -t
	
	sudo systemctl reload nginx
